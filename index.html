🔐 Google Apps Script Setup Guide (Token Security)
🛡️ Token Protection Rule:
Code ke andar line 7 par const SECRET_TOKEN = "APNA_CUSTOM_SECRET_TOKEN"; diya gaya hai. Yahan apna manchaaha secret password/token likhein, aur wahi exact token Web App ke Settings me Security Token box me enter karein.
Jab tak token match nahi hoga, Google Sheet se na data read (fetch) hoga aur na hi write (upload) ho sakega!

1. Apni Google Sheet open karein.
2. Menu me Extensions > Apps Script par click karein.
3. Purana code hata kar neeche diya gaya code paste karein.
4. Line 7 par SECRET_TOKEN ki jagah apna secret password likhein.
5. Deploy > New deployment > Select type: Web App par click karein.
6. Execute as: Me aur Who has access: Anyone select karke Deploy karein.
7. Web App URL ko copy karein aur Settings tab me paste karein.

Apps Script Code (With Strict Token Verification): 
// =========================================================================
// 🔐 GOOGLE APPS SCRIPT WITH TOKEN SECURITY FOR POINT & ACCOUNT LEDGER
// =========================================================================

// 🔴 APNA SECRET TOKEN YAHAN SET KAREIN (Change this to your strong password/token):
const SECRET_TOKEN = "MY_SECRET_TOKEN_123";

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return responseJSON({ status: "error", message: "No data received" });
    }

    var request = JSON.parse(e.postData.contents);
    var userToken = (request.token || "").toString().trim();

    // 🔒 Strictly Validate Token Before Allowing Any Changes
    if (!userToken || userToken !== SECRET_TOKEN) {
      return responseJSON({
        status: "error",
        message: "⛔ Access Denied: Invalid or missing Security Token!"
      });
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (request.action === "upload") {
      // 1. Entries Sheet
      var entrySheet = ss.getSheetByName("Entries") || ss.insertSheet("Entries");
      entrySheet.clearContents();
      entrySheet.appendRow([
        "Sr", "Date", "Account", "Pay In", "Pay Out", 
        "Opening Point Balance", "Closing Point Balance", "Points Profit", 
        "Ac Balance", "Commission", "Exp"
      ]);
      
      var entries = request.data.entries || [];
      for (var i = 0; i < entries.length; i++) {
        var row = entries[i];
        if (row.type === "transfer") {
          entrySheet.appendRow([
            i + 1, row.date, row.acName, "-", "-",
            "-", "-", "Transfer",
            row.acBalance, "-", "-"
          ]);
        } else {
          entrySheet.appendRow([
            i + 1, row.date, row.acName, row.payIn, row.payOut,
            row.openingPointBal, row.closingPointBal, row.pointsProfit,
            row.acBalance, row.commission, row.exp
          ]);
        }
      }
      
      // 2. Accounts Sheet
      var acSheet = ss.getSheetByName("Accounts") || ss.insertSheet("Accounts");
      acSheet.clearContents();
      acSheet.appendRow(["ID", "Account Name", "Current Balance"]);
      var accounts = request.data.accounts || [];
      for (var j = 0; j < accounts.length; j++) {
        acSheet.appendRow([accounts[j].id, accounts[j].name, accounts[j].balance]);
      }
      
      return responseJSON({
        status: "success",
        message: "Data uploaded successfully!",
        count: entries.length
      });
    }

    return responseJSON({ status: "error", message: "Unknown action requested." });

  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function doGet(e) {
  try {
    var userToken = (e && e.parameter && e.parameter.token ? e.parameter.token : "").toString().trim();

    // 🔒 Strictly Validate Token Before Allowing Any Data Read
    if (!userToken || userToken !== SECRET_TOKEN) {
      return responseJSON({
        status: "error",
        message: "⛔ Access Denied: Invalid or missing Security Token!"
      });
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var entrySheet = ss.getSheetByName("Entries");
    var entries = [];
    if (entrySheet) {
      var rows = entrySheet.getDataRange().getValues();
      for (var i = 1; i < rows.length; i++) {
        var r = rows[i];
        if (!r[1]) continue;
        var isTransfer = String(r[7]).toLowerCase().indexOf("transfer") !== -1;
        entries.push({
          id: "gs_" + i,
          type: isTransfer ? "transfer" : "entry",
          date: r[1],
          acName: r[2],
          payIn: isTransfer ? 0 : (Number(r[3]) || 0),
          payOut: isTransfer ? 0 : (Number(r[4]) || 0),
          openingPointBal: isTransfer ? 0 : (Number(r[5]) || 0),
          closingPointBal: isTransfer ? 0 : (Number(r[6]) || 0),
          pointsProfit: isTransfer ? 0 : (Number(r[7]) || 0),
          acBalance: r[8],
          commission: isTransfer ? 0 : (Number(r[9]) || 0),
          exp: isTransfer ? 0 : (Number(r[10]) || 0)
        });
      }
    }
    
    var acSheet = ss.getSheetByName("Accounts");
    var accounts = [];
    if (acSheet) {
      var aRows = acSheet.getDataRange().getValues();
      for (var j = 1; j < aRows.length; j++) {
        if (aRows[j][1]) {
          accounts.push({
            id: String(aRows[j][0]) || ("ac_" + j),
            name: aRows[j][1],
            balance: Number(aRows[j][2]) || 0
          });
        }
      }
    }

    return responseJSON({
      status: "success",
      data: { entries: entries, accounts: accounts }
    });

  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
